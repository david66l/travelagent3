                source_updated_at=datetime.now()

## Layer 2: 数据治理 & 自动更新

### 2.1 增量更新策略

```python
# backend/src/data/governance/updater.py

class IncrementalUpdater:
    """只更新变化字段，不重复覆盖静态数据"""
    
    async def sync_poi(self, raw: RawPOI):
        existing = await db.find_one("attractions", name=raw.name, city=raw.city)
        
        if not existing:
            # 新 POI → 完整入库
            await self._insert(raw)
            return
        
        # 已有 POI → 只更新动态字段
        changes = {}
        for field in ["ticket_price", "open_time", "close_time"]:
            if getattr(raw, field) != getattr(existing, field):
                changes[field] = getattr(raw, field)
        
        if changes:
            changes["source_updated_at"] = raw.source_updated_at
            await db.execute(
                "UPDATE attractions SET {} WHERE id = $1".format(
                    ", ".join(f"{k}=${i+2}" for i, k in enumerate(changes))
                ),
                existing.id, *changes.values()
            )
```

```python
# backend/src/data/governance/validator.py

class DataValidator:
    RULES = {
        "ticket_price": lambda v: v is None or v >= 0,
        "open_time": lambda v: v is None or re.match(r"\d{2}:\d{2}", v),
        "close_time": lambda v: v is None or re.match(r"\d{2}:\d{2}", v),
        "tuesday_closed": lambda poi: "周二" in (poi.close_time or "") or self._detect_weekly_close(poi, 1),
    }
    
    async def validate_all(self):
        suspicious = []
        for poi in await db.fetch_all("attractions"):
            errors = []
            for field, rule in self.RULES.items():
                if not rule(getattr(poi, field, None)):
                    errors.append(f"{field} invalid: {getattr(poi, field)}")
            if errors:
                suspicious.append({"poi_id": poi.id, "errors": errors})
        
        # LLM 二次校验矛盾数据
        for item in suspicious:
            await self._llm_verify(item)
        
        return suspicious

    # 过期淘汰
    async def mark_deprecated(self):
        """标记永久关闭的景区/酒店，规划时自动过滤"""
        await db.execute(
            "UPDATE attractions SET status='deprecated' "
            "WHERE source_updated_at < NOW() - INTERVAL '90 days' "
            "AND source = 'amap' AND amap_status = 'closed'"
        )
```

### Agent 查询接口

```python
# backend/src/data/repository.py

class TravelDataRepository:
    """Agent 唯一数据入口 — 只查本地库"""
    
    async def search_attractions(
        self, city: str, 
        filters: AttractionFilter  # 步行强度≤2, wheelchair=true, etc.
    ) -> list[Attraction]:
        """结构化过滤：毫秒级，不调外部 API"""
        query = "SELECT * FROM attractions WHERE city = $1 AND status != 'deprecated'"
        params = [city]
        
        if filters.max_walk_intensity:
            query += f" AND walk_intensity <= ${len(params)+1}"
            params.append(filters.max_walk_intensity)
        
        if filters.wheelchair_only:
            query += " AND wheelchair_accessible = true"
        
        if filters.avoid_tags:
            query += f" AND NOT (tags && ${len(params)+1}::text[])"
            params.append(filters.avoid_tags)
        
        return await db.fetch(query, *params)
    
    async def search_knowledge(
        self, query: str, city: str = None, top_k: int = 5
    ) -> list[KnowledgeTip]:
        """语义检索：向量 + 城市过滤"""
        embedding = await bge.encode(query)
        
        results = await db.fetch(
            """
            SELECT content, content_type, metadata,
                   1 - (embedding <=> $1) AS similarity
            FROM knowledge_tips
            WHERE ($2::text IS NULL OR city = $2)
            ORDER BY embedding <=> $1
            LIMIT $3
            """,
            embedding, city, top_k
        )
        return results
```

## Layer 4: 兜底实时检索

```python
# backend/src/data/fallback.py

class FallbackSearcher:
    """仅在本地库无结果时触发，结果写回本地库"""
    
    async def search(self, query: str) -> list[RawPOI]:
        # 1. 尝试本地库
        local = await repo.search_attractions(query)
        if local:
            return local
        
        # 2. 本地无 → 调高德 API
        logger.info("Local miss, falling back to Amap API")
        remote = await api_collector.search_amap(query)
        
        # 3. 写回本地库（下次直走本地）
        for poi in remote:
            await repo.upsert_attraction(poi)
        
        return remote
```
